Caesar Cipher

caesar_cipher.py
message = "HELLO WORLD"
shift = 3
clean_text = ""

for c in message:
    if c.isalpha():
        clean_text += c.upper()

encrypted_text = ""
for char in clean_text:
    val = ord(char) - 65
    new_val = (val + shift) % 26
    encrypted_text += chr(new_val + 65)
print(encrypted_text)

decrypted_text = ""
for char in encrypted_text:
    val = ord(char) - 65
    old_val = (val - shift) % 26
    decrypted_text += chr(old_val + 97)
print(decrypted_text)

Vigenère Cipher

vigenere_cipher.py
message = "ATTACK AT DAWN"
key = "LEMON"
clean_text = ""

for c in message:
    if c.isalpha():
        clean_text += c.upper()

extended_key = ""
key_index = 0
for i in range(len(clean_text)):
    extended_key += key[key_index % len(key)]
    key_index += 1

encrypted_text = ""
for i in range(len(clean_text)):
    p = ord(clean_text[i]) - 65
    k = ord(extended_key[i]) - 65
    c = (p + k) % 26
    encrypted_text += chr(c + 65)
print(encrypted_text)

decrypted_text = ""
for i in range(len(encrypted_text)):
    c = ord(encrypted_text[i]) - 65
    k = ord(extended_key[i]) - 65
    p = (c - k) % 26
    decrypted_text += chr(p + 97)
print(decrypted_text)

OTP (Sum & XOR)

otp_cipher.py
message = "HELLO"
key = "XMCKL"
clean_text = ""

for c in message:
    if c.isalpha():
        clean_text += c.upper()

xor_enc = []
for i in range(len(clean_text)):
    val = ord(clean_text[i]) ^ ord(key[i]) 
    xor_enc.append(val)
print(xor_enc)

xor_dec = ""
for i in range(len(clean_text)):
    val = xor_enc[i] ^ ord(key[i])
    xor_dec += chr(val).lower()
print(xor_dec)

sum_enc = []
for i in range(len(clean_text)):
    val = ord(clean_text[i]) + ord(key[i])
    sum_enc.append(val)
print(sum_enc)

sum_dec = ""
for i in range(len(clean_text)):
    val = sum_enc[i] - ord(key[i])
    sum_dec += chr(val).lower()
print(sum_dec)

Affine Cipher

affine_cipher.py
message = "AFFINE"
a = 5
b = 8
clean_text = ""

for c in message:
    if c.isalpha():
        clean_text += c.upper()

encrypted_text = ""
for char in clean_text:
    x = ord(char) - 65
    e = (a * x + b) % 26
    encrypted_text += chr(e + 65)
print(encrypted_text)

decrypted_text = ""
a_inv = 0
for i in range(26):
    if (a * i) % 26 == 1:
        a_inv = i

for char in encrypted_text:
    e = ord(char) - 65
    d = (a_inv * (e - b)) % 26
    decrypted_text += chr(d + 97)
print(decrypted_text)

Frequency Analysis

frequency_analysis.py
text = "THE QUICK BROWN FOX JUMPS OVER THE LAZY DOG"
clean_text = ""

for c in text:
    if c.isalpha():
        clean_text += c.upper()

frequencies = {}
for char in clean_text:
    if char in frequencies:
        frequencies[char] += 1
    else:
        frequencies[char] = 1

alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
for letter in alphabet:
    if letter in frequencies:
        print(letter, ":", frequencies[letter])

RSA Encryption

rsa_simple.py
p = 11
q = 3
n = p * q
phi = (p - 1) * (q - 1)
e = 3
d = 0

for k in range(100):
    if (k * e) % phi == 1:
        d = k
        break

print(e, n)
print(d, n)

original = 7
print(original)

encrypted = (original ** e) % n
print(encrypted)

decrypted = (encrypted ** d) % n
