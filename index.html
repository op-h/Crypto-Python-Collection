Remove Characters

Removes a specific letter from a string without using libraries or functions.

remove_characters.py
text = "Hello World, this is a python code."
character_to_remove = "o"
result = ""

print("Original Text:", text)
print("Character to remove:", character_to_remove)

for char in text:
    if char != character_to_remove:
        result = result + char

print("Result:", result)

Caesar Cipher

Shifts letters by a fixed number (Key=3) to encrypt and decrypt.

caesar_cipher.py
message = "HELLO WORLD"
shift = 3
encrypted_text = ""
decrypted_text = ""

print("Original Message:", message)
print("Shift Key:", shift)

for char in message:
    if char.isalpha():
        original_position = ord(char) - 65
        new_position = (original_position + shift) % 26
        new_char = chr(new_position + 65)
        encrypted_text += new_char
    else:
        encrypted_text += char

print("Encrypted:", encrypted_text)

for char in encrypted_text:
    if char.isalpha():
        current_position = ord(char) - 65
        original_position = (current_position - shift) % 26
        original_char = chr(original_position + 65)
        decrypted_text += original_char
    else:
        decrypted_text += char

print("Decrypted:", decrypted_text)

Substitution Cipher

Replaces every letter based on a scrambled alphabet key.

substitution_cipher.py
plain_alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
cipher_alphabet = "QWERTYUIOPASDFGHJKLZXCVBNM"

message = "HELLO PYTHON"
encrypted_text = ""
decrypted_text = ""

print("Original:", message)

for char in message:
    if char in plain_alphabet:
        index = plain_alphabet.find(char)
        encrypted_text += cipher_alphabet[index]
    else:
        encrypted_text += char

print("Encrypted:", encrypted_text)

for char in encrypted_text:
    if char in cipher_alphabet:
        index = cipher_alphabet.find(char)
        decrypted_text += plain_alphabet[index]
    else:
        decrypted_text += char

print("Decrypted:", decrypted_text)

Vigenère Cipher

Encryption using a keyword to shift letters variably.

vigenere_cipher.py
message = "ATTACKATDAWN"
key = "LEMON"
encrypted_text = ""
decrypted_text = ""
extended_key = ""
key_index = 0

for char in message:
    if char.isalpha():
        extended_key += key[key_index % len(key)]
        key_index += 1
    else:
        extended_key += char

print("Message:", message)
print("Key:    ", extended_key)

for i in range(len(message)):
    char = message[i]
    if char.isalpha():
        p_val = ord(char) - 65
        k_val = ord(extended_key[i]) - 65
        c_val = (p_val + k_val) % 26
        encrypted_text += chr(c_val + 65)
    else:
        encrypted_text += char

print("Encrypted:", encrypted_text)

for i in range(len(encrypted_text)):
    char = encrypted_text[i]
    if char.isalpha():
        c_val = ord(char) - 65
        k_val = ord(extended_key[i]) - 65
        p_val = (c_val - k_val + 26) % 26
        decrypted_text += chr(p_val + 65)
    else:
        decrypted_text += char

print("Decrypted:", decrypted_text)

OTP (Sum & XOR)

Demonstrates encryption using both bitwise XOR and Simple Sum.

otp_cipher.py
message = "HELLO"
key = "XMCKL"

print("Message:", message)
print("Key:    ", key)

xor_encrypted = []
for i in range(len(message)):
    val = ord(message[i]) ^ ord(key[i]) 
    xor_encrypted.append(val)

print("Encrypted (numbers):", xor_encrypted)

xor_decrypted = ""
for i in range(len(message)):
    val = xor_encrypted[i] ^ ord(key[i])
    xor_decrypted += chr(val)

print("Decrypted:", xor_decrypted)

sum_encrypted = []
for i in range(len(message)):
    val = ord(message[i]) + ord(key[i])
    sum_encrypted.append(val)

print("Encrypted (numbers):", sum_encrypted)

sum_decrypted = ""
for i in range(len(message)):
    val = sum_encrypted[i] - ord(key[i])
    sum_decrypted += chr(val)

print("Decrypted:", sum_decrypted)

Affine Cipher

Mathematical encryption using formula (ax + b) % 26.

affine_cipher.py
message = "AFFINE"
a = 5
b = 8
encrypted_text = ""
decrypted_text = ""

print("Message:", message)

for char in message:
    if char.isalpha():
        x = ord(char) - 65
        e = (a * x + b) % 26
        encrypted_text += chr(e + 65)
    else:
        encrypted_text += char

print("Encrypted:", encrypted_text)

a_inv = 0
for i in range(26):
    if (a * i) % 26 == 1:
        a_inv = i
        break

if a_inv != 0:
    for char in encrypted_text:
        if char.isalpha():
            e = ord(char) - 65
            d = (a_inv * (e - b)) % 26
            decrypted_text += chr(d + 65)
        else:
            decrypted_text += char
    print("Decrypted:", decrypted_text)
else:
    print("Error: 'a' has no modular inverse.")

Frequency Analysis

Counts occurrences of letters in a text using a dictionary.

frequency_analysis.py
text = "THE QUICK BROWN FOX JUMPS OVER THE LAZY DOG"
print("Text:", text)

frequencies = {}

for char in text:
    if char.isalpha():
        if char in frequencies:
            frequencies[char] = frequencies[char] + 1
        else:
            frequencies[char] = 1

alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

for letter in alphabet:
    if letter in frequencies:
        count = frequencies[letter]
        print(letter, ":", count)

RSA Encryption (Simplified)

Basic RSA mathematics showing key generation and exponential math.

rsa_simple.py
p = 11
q = 3
n = p * q
phi = (p - 1) * (q - 1)
e = 3
d = 0

for k in range(100):
    if (k * e) % phi == 1:
        d = k
        break

print("Public Key: (e=", e, ", n=", n, ")")
print("Private Key: (d=", d, ", n=", n, ")")

original_number = 7
print("Original Number:", original_number)

encrypted_number = (original_number ** e) % n
print("Encrypted Number:", encrypted_number)

decrypted_number = (encrypted_number ** d) % n
print("Decrypted Number:", decrypted_number)

GCD (Greatest Common Divisor)

Finds the GCD of two numbers using Euclidean algorithm loops.

gcd_simple.py
num1 = 48
num2 = 18

print("Number 1:", num1)
print("Number 2:", num2)

while num2 != 0:
    temp = num2
    num2 = num1 % num2
    num1 = temp

print("GCD is:", num1)
